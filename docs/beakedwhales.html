# sample 10k clicks
set.seed(1407)
click_sample <- slice_sample(clicks, n = 1e4) %>% 
  rbind(archetypes)

#normalize everything by z score (sam did along 0-1)
z_score <- function(x) {
  stopifnot(sum(!is.na(x)) > 3)
  (x - mean(x, na.rm = TRUE)) / sd(x, na.rm = TRUE)
}
click_norm <- mutate(click_sample, across(everything(), z_score))

HDBSCAN is a neighborhood based clustering algorithm. It takes one parameter, minPts, which sets a minimum for cluster size. Let’s set minPts to 100.
