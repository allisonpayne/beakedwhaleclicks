clicks <- arrow::read_feather("data/features_20190529.feather") 
#every feather file holds a day

#transforming duration to log(duration) and selecting the four features that Sam used to cluster
clicks <- clicks %>% 
  mutate(logduration = log(duration)) %>% 
  select(logduration, peak_freq, peak_bw, freq50)

#subsample every n clicks
click_idx <- sample(1:nrow(clicks), 22200)
click_sub <- clicks[click_idx, ]

#normalize everything along 0-1
click_norm_sub <- (click_sub - min(click_sub)) / (max(click_sub) - min(click_sub))
